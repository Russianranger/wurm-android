cmake_minimum_required(VERSION 3.22)
project(WurmAndroidAudio LANGUAGES C CXX)

if(NOT ANDROID OR NOT ANDROID_ABI STREQUAL "arm64-v8a")
    message(FATAL_ERROR "This audio package requires Android arm64-v8a")
endif()
if(NOT WURM_OPENAL_SOURCE)
    message(FATAL_ERROR "Set WURM_OPENAL_SOURCE to the verified OpenAL Soft archive")
endif()

set(ALSOFT_UTILS OFF CACHE BOOL "" FORCE)
set(ALSOFT_EXAMPLES OFF CACHE BOOL "" FORCE)
set(ALSOFT_TESTS OFF CACHE BOOL "" FORCE)
set(ALSOFT_INSTALL OFF CACHE BOOL "" FORCE)
set(ALSOFT_UPDATE_BUILD_VERSION OFF CACHE BOOL "" FORCE)
foreach(backend ALSA OSS PIPEWIRE PULSEAUDIO JACK PORTAUDIO COREAUDIO SNDIO SOLARIS OBOE WAVE SDL2 SDL3)
    set(ALSOFT_BACKEND_${backend} OFF CACHE BOOL "" FORCE)
endforeach()
set(ALSOFT_BACKEND_OPENSL ON CACHE BOOL "" FORCE)
set(ALSOFT_REQUIRE_OPENSL ON CACHE BOOL "" FORCE)
add_subdirectory("${WURM_OPENAL_SOURCE}" openal)
set_target_properties(OpenAL PROPERTIES OUTPUT_NAME wurm_openal)
target_link_options(OpenAL PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,--no-undefined")
